// Load TerraClimate dataset
var dataset = ee.ImageCollection('IDAHO_EPSCOR/TERRACLIMATE')
                .filterBounds(roi); // Filter by ROI

// Function to calculate annual sum of aet
var calculateAnnualAETSum = function(year) {
  var startDate = ee.Date.fromYMD(year, 1, 1);
  var endDate = startDate.advance(1, 'year');
  
  // Filter dataset for the year and calculate the sum
  var annualAETSum = dataset.filterDate(startDate, endDate)
                            .select('aet') // Select aet band
                            .sum() // Calculate annual sum
                            .multiply(0.1) // Apply scaling factor
                            .clip(roi) // Clip to ROI
                            .set('year', year) // Add year metadata
                            .set('system:time_start', startDate.millis()); // Set time metadata
  
  return annualAETSum;
};

// Generate a collection of annual aet sum from 2001 to 2020
var years = ee.List.sequence(2001, 2020);
var annualAETCollection = ee.ImageCollection.fromImages(
  years.map(calculateAnnualAETSum)
);

// Visualization parameters
var aetVis = {

  max: 1000, // Adjust based on your region's data range

    'ffffff', 'c6e2ff', 'a5c9ff', '6499f3', '3a77c0', '1f57a5',
    '003b5c', '001f3b'
  ],
};

Map.centerObject(roi, 6);

var calculateAnnualAverageLST = function(year) {
  var startDate = ee.Date.fromYMD(year, 1, 1);
  var endDate = startDate.advance(1, 'year');
  
  // Filter dataset for the year and apply scaling
  var annualLST = dataset.filterDate(startDate, endDate)
                         .select('LST_Day_1km')

                           return image.multiply(0.02).subtract(273.15).set('system:time_start', image.get('system:time_start')); // Apply scaling factor and convert to Celsius
                         })
                         .mean() // Calculate annual average
                         .set('year', year)
                         .clip(roi); // Clip to the ROI
  
  return annualLST.set('system:time_start', startDate.millis());
};

// ============================================================
// MSAVI ANNUAL ANALYSIS: SUNDARBANS
// Study period: 2001–2020
// Spatial resolution: 30 m
// Datasets: Landsat 5 TM + Landsat 8 OLI
// Output: Annual MSAVI rasters + CSV summary
// ============================================================


// ------------------------------------------------------------
// 1. FUNCTION TO MASK CLOUDS, SHADOWS AND SNOW
// ------------------------------------------------------------

function maskLandsatC2(image) {
  var qa = image.select('QA_PIXEL');

  var mask = qa.bitwiseAnd(1 << 1).eq(0)  // Dilated cloud
    .and(qa.bitwiseAnd(1 << 2).eq(0))     // Cirrus
    .and(qa.bitwiseAnd(1 << 3).eq(0))     // Cloud
    .and(qa.bitwiseAnd(1 << 4).eq(0))     // Cloud shadow
    .and(qa.bitwiseAnd(1 << 5).eq(0));    // Snow

  return image.updateMask(mask);
}


// ------------------------------------------------------------
// 2. FUNCTION TO CALCULATE MSAVI
// Standard formula:
// MSAVI = [2*NIR + 1 - sqrt((2*NIR + 1)^2 - 8*(NIR-RED))] / 2
// ------------------------------------------------------------

function calculateMSAVI(image, nirBand, redBand) {

  var scaled = image.select([nirBand, redBand])
    .multiply(0.0000275)
    .add(-0.2);

  var nir = scaled.select(nirBand);
  var red = scaled.select(redBand);

  var discriminant = nir.multiply(2).add(1).pow(2)
    .subtract(nir.subtract(red).multiply(8))
    .max(0);

  var msavi = nir.multiply(2).add(1)
    .subtract(discriminant.sqrt())
    .divide(2)
    .rename('MSAVI');

  return msavi.copyProperties(image, [
    'system:time_start',
    'LANDSAT_PRODUCT_ID'
  ]);
}


// ------------------------------------------------------------
// 3. PROCESS LANDSAT 5 TM
// ------------------------------------------------------------

var landsat5 = ee.ImageCollection('LANDSAT/LT05/C02/T1_L2')
  .filterBounds(roi)
  .filterDate('2001-01-01', '2012-01-01')
  .map(maskLandsatC2)
  .map(function(image) {
    return calculateMSAVI(image, 'SR_B4', 'SR_B3');
  });


// ------------------------------------------------------------
// 4. PROCESS LANDSAT 8 OLI
// ------------------------------------------------------------

var landsat8 = ee.ImageCollection('LANDSAT/LC08/C02/T1_L2')
  .filterBounds(roi)
  .filterDate('2013-01-01', '2021-01-01')
  .map(maskLandsatC2)
  .map(function(image) {
    return calculateMSAVI(image, 'SR_B5', 'SR_B4');
  });


// ------------------------------------------------------------
// 5. MERGE COLLECTIONS
// ------------------------------------------------------------

var mergedCollection = landsat5.merge(landsat8);

print('Merged MSAVI collection:', mergedCollection);


// ------------------------------------------------------------
// 6. DEFINE STUDY YEARS
// ------------------------------------------------------------

var years = ee.List.sequence(2001, 2020);


// ------------------------------------------------------------
// 7. CALCULATE ANNUAL MEAN MSAVI AND OBSERVATION COUNT
// ------------------------------------------------------------

var annualMSAVI = ee.ImageCollection.fromImages(
  years.map(function(year) {

    year = ee.Number(year);

    var startDate = ee.Date.fromYMD(year, 1, 1);
    var endDate = startDate.advance(1, 'year');

    var yearlyCollection = mergedCollection.filterDate(
      startDate, endDate
    );

    var annualMean = yearlyCollection.mean()
      .rename('MSAVI')
      .clip(roi);

    var observationCount = yearlyCollection.count()
      .rename('Observation_Count')
      .clip(roi);

    return annualMean.addBands(observationCount)
      .set('year', year)
      .set('system:time_start', startDate.millis())
      .set('image_count', yearlyCollection.size());
  })
);

print('Annual MSAVI collection (2001–2020):', annualMSAVI);


// ------------------------------------------------------------
// 8. VISUALIZE MSAVI FOR 2020
// ------------------------------------------------------------

var msavi2020 = annualMSAVI
  .filter(ee.Filter.eq('year', 2020))

Code for mann kendall test and senslope
LAI
MODIS LAI TREND ANALYSIS: SUNDARBANS
 Period: 2001–2020
 Dataset: MODIS/061/MOD15A2H
 Nominal spatial resolution: 500 m

 PANEL A: Theil–Sen slope (LAI units/year)
 PANEL B: Mann–Kendall trend significance classification
 PANEL C: Binary significance mask

 Outputs:
 1. Theil–Sen slope
 2. Mann–Kendall S statistic
 3. Approximate standardized Z statistic
 4. Approximate two-sided p-value
 5. Trend classification
 6. Binary significance mask
 7. Valid annual observation count
 8. CSV summary of pixel counts and percentages

 Statistical limitations:
 - Approximate MK variance without tie correction
 - No correction for temporal autocorrelation
 - Minimum 18 valid annual observations
***************************************************************/

// ============================================================
// 1. STUDY AREA AND PARAMETERS
// ============================================================

// Import the Sundarbans boundary as "roi".
// If your imported geometry is named "geometry", use:
// var roi = geometry;

var START_YEAR = 2001;
var END_YEAR = 2020;
var ALPHA = 0.05;
var MIN_OBS = 18;
var SCALE = 500;

Map.centerObject(roi, 7);

// ============================================================
// 2. LOAD MODIS LAI AND QUALITY SCREENING
// ============================================================

var modis = ee.ImageCollection('MODIS/061/MOD15A2H')
  .filterBounds(roi)
  .filterDate(
    ee.Date.fromYMD(START_YEAR, 1, 1),
    ee.Date.fromYMD(END_YEAR + 1, 1, 1)
  )
  .select(['Lai_500m', 'FparLai_QC']);

var laiCollection = modis.map(function(img) {
  var raw = img.select('Lai_500m');
  var qc = img.select('FparLai_QC');

  // MODLAND QC bits 0–1 = 00: good-quality retrieval.
  var goodQuality = qc.bitwiseAnd(3).eq(0);

  // LAI scale factor = 0.1.
  // Mask fill and out-of-range raw values.
  var valid = raw.gte(0).and(raw.lte(100));

  var lai = raw.multiply(0.1)
    .rename('LAI')
    .updateMask(goodQuality)
    .updateMask(valid);

  return lai.copyProperties(img, ['system:time_start']);
});

// ============================================================
// 3. ANNUAL MEAN LAI, 2001–2020
// ============================================================

var years = ee.List.sequence(START_YEAR, END_YEAR);

var annualLAI = ee.ImageCollection.fromImages(
  years.map(function(y) {
    y = ee.Number(y);

    var start = ee.Date.fromYMD(y, 1, 1);
    var end = start.advance(1, 'year');

    var annual = laiCollection
      .filterDate(start, end)
      .mean()
      .rename('LAI')
      .clip(roi);

    return annual
      .set('year', y)
      .set('system:time_start', start.millis());
  })
).sort('system:time_start');

print('Annual LAI collection', annualLAI);
print('Number of annual images', annualLAI.size());

// Count valid annual LAI observations at each pixel.
var n = annualLAI.select('LAI')
  .count()
  .rename('N_Valid')
  .clip(roi);

var validPeriod = n.gte(MIN_OBS);

Map.addLayer(
  n,
  {
    min: 0,
    max: 20,
    palette: ['FFFFFF', '74ADD1', '2166AC']
  },
  'Valid annual observations',
  false
);

// ============================================================
// 4. THEIL–SEN SLOPE: TREND MAGNITUDE
// ============================================================

var yearArray = [];
for (var y = START_YEAR; y <= END_YEAR; y++) {
  yearArray.push(y);
}

var slopeImages = [];
var signImages = [];

for (var i = 0; i < yearArray.length; i++) {
  for (var j = i + 1; j < yearArray.length; j++) {

    var year1 = yearArray[i];
    var year2 = yearArray[j];

    var img1 = ee.Image(
      annualLAI.filter(ee.Filter.eq('year', year1)).first()
    ).select('LAI');

    var img2 = ee.Image(
      annualLAI.filter(ee.Filter.eq('year', year2)).first()
    ).select('LAI');

    var difference = img2.subtract(img1);
    var elapsed = year2 - year1;

    // Pairwise slope in LAI units/year.
    var pairSlope = difference
      .divide(elapsed)
      .rename('slope');

    slopeImages.push(pairSlope);

    // Mann–Kendall pairwise sign.
    var pairSign = difference.gt(0)
      .subtract(difference.lt(0))
      .rename('sign');

    signImages.push(pairSign);
  }
}

var slopeCollection = ee.ImageCollection.fromImages(slopeImages);

var senSlope = slopeCollection
  .median()
  .rename('Sen_Slope')
  .updateMask(validPeriod)
  .clip(roi);

// ============================================================
// 5. MANN–KENDALL S STATISTIC
// ============================================================

var mkS = ee.ImageCollection.fromImages(signImages)
  .sum()
  .rename('MK_S')
  .updateMask(validPeriod)
  .clip(roi);

// ============================================================
// 6. APPROXIMATE MK VARIANCE, Z AND P
// ============================================================

// No-tie-correction variance:
// Var(S) = n(n-1)(2n+5)/18
//
// n is calculated separately for every pixel.
// This is an approximate test; ties and serial correlation
// require additional consideration for final inference.

var variance = n
  .multiply(n.subtract(1))
  .multiply(n.multiply(2).add(5))
  .divide(18)
  .updateMask(validPeriod);

// Continuity-corrected standardized Z statistic.
var z = mkS.expression(
  "(S > 0) ? (S - 1) / sqrt(V) : " +
  "((S < 0) ? (S + 1) / sqrt(V) : 0)",
  {
    S: mkS,
    V: variance
  }
)
.rename('MK_Z')
.updateMask(validPeriod);

// Approximate two-sided p-value under the standard normal model.
var pValue = ee.Image(1)
  .subtract(
    z.abs()
      .divide(Math.sqrt(2))
      .erf()
  )
  .rename('MK_p')
  .clamp(0, 1)
  .updateMask(validPeriod)
  .clip(roi);

// ============================================================
// 7. TREND CLASS AND SIGNIFICANCE MASK
// ============================================================

// Trend_Class:
// 0 = zero slope
// 1 = significant increase
// 2 = significant decrease
// 3 = non-significant increase
// 4 = non-significant decrease

var trendClass = ee.Image(0)
  .where(senSlope.gt(0).and(pValue.lt(ALPHA)), 1)
  .where(senSlope.lt(0).and(pValue.lt(ALPHA)), 2)
  .where(senSlope.gt(0).and(pValue.gte(ALPHA)), 3)
  .where(senSlope.lt(0).and(pValue.gte(ALPHA)), 4)
  .rename('Trend_Class')
  .toInt8()
  .updateMask(validPeriod)
  .clip(roi);

// Binary significance mask:
// 1 = statistically significant at alpha = 0.05
// 0 = not statistically significant
// Pixels with insufficient observations remain masked.

var significanceMask = pValue.lt(ALPHA)
  .rename('Significant_Mask')
  .toByte()
  .updateMask(validPeriod)
  .clip(roi);

// Combine all output bands.
var results = senSlope
  .addBands(mkS)
  .addBands(z)
  .addBands(pValue)
  .addBands(n)
  .addBands(trendClass)
  .addBands(significanceMask)
  .clip(roi);

print('Final LAI trend results', results);
print('Output band names', results.bandNames());

var validPixelImage = ee.Image.constant(1)

// 3. CALCULATE ANNUAL PRECIPITATION
// ============================================================

// Sum the monthly values to obtain annual precipitation in mm/year.

for (var y = START_YEAR; y <= END_YEAR; y++) {
  yearList.push(y);
}

var annualImages = yearList.map(function(year) {

  var start = ee.Date.fromYMD(year, 1, 1);

  var annual = monthly
    .select('pr')

  .select(['EVI', 'SummaryQA']);

  var rawEVI = img.select('EVI');
  var qa = img.select('SummaryQA');

  // SummaryQA:
  // 0 = good quality

  var qualityMask = qa.lte(1);

  // Exclude fill values and invalid scaled values.
  var validRange = rawEVI.gte(-2000)
    .and(rawEVI.lte(10000));

    .multiply(0.0001)
    .rename('EVI')

// ============================================================
// MODIS DAYTIME LAND SURFACE TEMPERATURE TREND ANALYSIS
// Study period: 2001–2020
// Dataset: MODIS/061/MOD11A1
// Nominal spatial resolution: 1 km
// Variable: Annual mean daytime LST (°C)
//
// OUTPUTS:
// 1. Annual LST stack
// 2. Theil–Sen slope (°C/year)
// 3. Mann–Kendall S statistic
// 4. Approximate MK Z statistic
// 5. Approximate two-sided MK p-value
// 6. Trend classification
// 7. Significance mask
// 8. Valid annual observation count
// 9. CSV pixel summary
// 10. Combined and individual GeoTIFFs
//
// IMPORTANT:
// - Requires an imported study-area geometry named "roi".
// - MK variance is approximate; tie and autocorrelation
//   corrections are not included.
// - Missing-year handling should be validated before
//   interpreting pixel-level significance.
// ============================================================


// ------------------------------------------------------------
// 1. SETTINGS
// ------------------------------------------------------------

var START_YEAR = 2001;
var END_YEAR = 2020;
var ALPHA = 0.05;
var MIN_OBS = 18;

var SCALE = 1000;
var EXPORT_CRS = 'EPSG:32645';

Map.centerObject(roi, 6);


// ------------------------------------------------------------
// 2. LOAD DAILY MODIS DATA
// Avoid printing the full daily ImageCollection.
// ------------------------------------------------------------

var source = ee.ImageCollection('MODIS/061/MOD11A1')
  .filterBounds(roi)
  .filterDate(
    ee.Date.fromYMD(START_YEAR, 1, 1),
    ee.Date.fromYMD(END_YEAR + 1, 1, 1)
  )
  .select(['LST_Day_1km', 'QC_Day']);

print('Daily image count', source.size());


// ------------------------------------------------------------
// 3. QUALITY CONTROL AND CONVERSION TO CELSIUS
// ------------------------------------------------------------
// MOD11A1 LST scale factor = 0.02 Kelvin
// LST (°C) = raw LST × 0.02 - 273.15
//
// QC_Day bits:
// 0–1: mandatory QA, 00 = good
// 2–3: data quality, 00 = good
// 6–7: LST error class, 0 or 1
// ------------------------------------------------------------

function prepareLST(image) {

  var raw = image.select('LST_Day_1km');
  var qc = image.select('QC_Day');

  var mandatoryQA = qc.bitwiseAnd(3).eq(0);

  var dataQA = qc
    .rightShift(2)
    .bitwiseAnd(3)
    .eq(0);

  var errorQA = qc
    .rightShift(6)
    .bitwiseAnd(3)
    .lte(1);

  var valid = raw.gt(0)
    .and(mandatoryQA)
    .and(dataQA)
    .and(errorQA);

  var lstCelsius = raw
    .multiply(0.02)
    .subtract(273.15)
    .rename('LST')
    .updateMask(valid)
    .toFloat();

  return lstCelsius.copyProperties(
    image, ['system:time_start']
  );
}


// ------------------------------------------------------------
// 4. CREATE ANNUAL MEAN LST COMPOSITES
// All annual images must have the SAME band name: "LST".
// ------------------------------------------------------------

var years = [];
var annualImages = [];

for (var year = START_YEAR; year <= END_YEAR; year++) {

  years.push(year);

  var startDate = ee.Date.fromYMD(year, 1, 1);
  var endDate = startDate.advance(1, 'year');

  var annualCollection = source
    .filterDate(startDate, endDate)
    .map(prepareLST);

  var annualImage = annualCollection
    .mean()
    .rename('LST')
    .toFloat()
    .clip(roi)
    .set('year', year)
    .set('system:time_start', startDate.millis());

  annualImages.push(annualImage);
}

var annualLST = ee.ImageCollection
  .fromImages(annualImages)
  .sort('system:time_start');

print('Annual composite count', annualLST.size());


// ------------------------------------------------------------
// 5. STACK ANNUAL IMAGES
// Rename bands only AFTER creating the ImageCollection.
// ------------------------------------------------------------

var stackedLST = annualLST.toBands();

var bandNames = years.map(function(year) {
  return 'LST_' + year;
});

stackedLST = stackedLST
  .rename(bandNames)
  .toFloat();

print('Annual LST band names', stackedLST.bandNames());


// ------------------------------------------------------------
// 6. VALID ANNUAL OBSERVATION COUNT
// count() is an ImageCollection method.
// ------------------------------------------------------------

var validYearCount = annualLST
  .count()
  .rename('Valid_Years')
  .clip(roi);

var validMask = validYearCount.gte(MIN_OBS);

Map.addLayer(
  validYearCount,
  {
    min: 0,
    max: 20,
    palette: [
      '8c2d04', 'fecc5c', 'ffffb2', '31a354'
    ]
  },
  'Valid annual observations',
  false
);


// ------------------------------------------------------------
// 7. DISPLAY ANNUAL LST FOR 2010
// ------------------------------------------------------------

Map.addLayer(
  stackedLST
    .select('LST_2010')
    .updateMask(validMask),
  {
    min: 0,
    max: 42,
    palette: [
      '313695', '4575b4', '74add1', 'abd9e9',
      'e0f3f8', 'ffffbf', 'fee090', 'fdae61',
      'f46d43', 'd73027', 'a50026'

  },

// ------------------------------------------------------------
// 8. THEIL–SEN SLOPE
// Median of pairwise annual slopes.
// Units: °C/year
// ------------------------------------------------------------

var pairwiseSlopes = [];

for (var i = 0; i < years.length; i++) {

  for (var j = i + 1; j < years.length; j++) {

    var yearI = years[i];
    var yearJ = years[j];

    var imageI = stackedLST.select('LST_' + yearI);
    var imageJ = stackedLST.select('LST_' + yearJ);

    var pairSlope = imageJ
      .subtract(imageI)
      .divide(yearJ - yearI)
      .rename('Pairwise_Slope')
      .toFloat();

    pairwiseSlopes.push(pairSlope);
  }
}

var senSlope = ee.ImageCollection
  .fromImages(pairwiseSlopes)
  .median()
  .rename('Theil_Sen_Slope')
  .toFloat()
  .updateMask(validMask)
  .clip(roi);

print('Theil–Sen slope image', senSlope);


// ------------------------------------------------------------
// 9. MANN–KENDALL S STATISTIC
// S = sum of sign(Xj - Xi) for all j > i
//
// Positive S = warming tendency
// Negative S = cooling tendency
// S = 0 = no net direction
//
// This is conventional, unnormalized S.
// ------------------------------------------------------------

var pairwiseSigns = [];

for (var i = 0; i < years.length; i++) {

  for (var j = i + 1; j < years.length; j++) {

    var xi = stackedLST.select('LST_' + years[i]);
    var xj = stackedLST.select('LST_' + years[j]);

    var difference = xj.subtract(xi);

    var sign = difference.gt(0)
      .subtract(difference.lt(0))
      .rename('Sign')
      .unmask(0)
      .toFloat();

    pairwiseSigns.push(sign);
  }
}

var mkS = ee.ImageCollection
  .fromImages(pairwiseSigns)
  .sum()
  .rename('MK_S')
  .toFloat()
  .updateMask(validMask)
  .clip(roi);

print('Mann–Kendall S statistic', mkS);


// ------------------------------------------------------------
// 10. APPROXIMATE MK Z STATISTIC AND P-VALUE
// ------------------------------------------------------------
// Var(S) = n(n - 1)(2n + 5) / 18
//
// Continuity correction:
// S > 0: S - 1
// S < 0: S + 1
// S = 0: 0
//
// p = 1 - erf(|Z| / sqrt(2))
//
// This approximation does not apply tie or serial-correlation
// correction.
// ------------------------------------------------------------

var n = validYearCount.toFloat();

var varianceS = n
  .multiply(n.subtract(1))
  .multiply(n.multiply(2).add(5))
  .divide(18)
  .toFloat();

var zNumerator = ee.Image.constant(0).toFloat()
  .where(mkS.gt(0), mkS.subtract(1))
  .where(mkS.lt(0), mkS.add(1));

var mkZ = zNumerator
  .divide(varianceS.sqrt())
  .rename('MK_Z')
  .toFloat()
  .updateMask(validMask)
  .clip(roi);

var mkP = ee.Image.constant(1).toFloat()
  .subtract(
    mkZ.abs()
      .divide(Math.sqrt(2))
      .erf()
  )
  .clamp(0, 1)
  .rename('MK_p')
  .toFloat()
  .updateMask(validMask)
  .clip(roi);

print('MK Z statistic', mkZ);
print('Approximate MK p-value', mkP);


// ------------------------------------------------------------
// 11. TREND CLASSIFICATION
// 0 = zero slope
// 1 = significant warming
// 2 = significant cooling
// 3 = non-significant warming
// 4 = non-significant cooling
// ------------------------------------------------------------

var trendClass = ee.Image.constant(0)
  .toByte()
  .where(senSlope.gt(0).and(mkP.lt(ALPHA)), 1)
  .where(senSlope.lt(0).and(mkP.lt(ALPHA)), 2)
  .where(senSlope.gt(0).and(mkP.gte(ALPHA)), 3)
  .where(senSlope.lt(0).and(mkP.gte(ALPHA)), 4)
  .rename('Trend_Class')
  .updateMask(validMask)
  .clip(roi);

var significanceMask = mkP.lt(ALPHA)
  .rename('Significant_Trend')
  .toByte()
  .updateMask(validMask)
  .clip(roi);

// ------------------------------------------------------------
// 2. CLOUD, SHADOW AND SATURATION MASK
// Landsat Collection 2 Level 2
// ------------------------------------------------------------

function maskLandsat(image) {
  var qa = image.select('QA_PIXEL');
  var saturation = image.select('QA_RADSAT');

  // QA_PIXEL bits:
  // 0 fill, 1 dilated cloud, 2 cirrus,
  // 3 cloud, 4 cloud shadow, 5 snow
  var mask = qa.bitwiseAnd(1 << 0).eq(0)
    .and(qa.bitwiseAnd(1 << 1).eq(0))
    .and(qa.bitwiseAnd(1 << 2).eq(0))
    .and(qa.bitwiseAnd(1 << 3).eq(0))
    .and(qa.bitwiseAnd(1 << 4).eq(0))
    .and(qa.bitwiseAnd(1 << 5).eq(0))
    .and(saturation.eq(0));

  return image.updateMask(mask);
}


// ------------------------------------------------------------
// 3. LANDSAT 5 MNDWI
// Green = SR_B2
// SWIR1 = SR_B5
// ------------------------------------------------------------

function calculateMNDWI_L5(image) {
  image = maskLandsat(image);

  var green = image.select('SR_B2')
    .multiply(0.0000275)
    .add(-0.2);

  var swir1 = image.select('SR_B5')
    .multiply(0.0000275)
    .add(-0.2);

  var mndwi = green.subtract(swir1)
    .divide(green.add(swir1))
    .rename('MNDWI');

  // Mask zero/near-zero denominators.
  mndwi = mndwi.updateMask(
    green.add(swir1).abs().gt(0.0001)
  );

  return mndwi
    .toFloat()
    .copyProperties(image, ['system:time_start']);
}


// ------------------------------------------------------------
// 4. LANDSAT 8 MNDWI
// Green = SR_B3
// SWIR1 = SR_B6
// ------------------------------------------------------------

function calculateMNDWI_L8(image) {
  image = maskLandsat(image);

  var green = image.select('SR_B3')
    .multiply(0.0000275)
    .add(-0.2);

  var swir1 = image.select('SR_B6')
    .multiply(0.0000275)
    .add(-0.2);

  var mndwi = green.subtract(swir1)
    .divide(green.add(swir1))
    .rename('MNDWI');

  mndwi = mndwi.updateMask(
    green.add(swir1).abs().gt(0.0001)
  );

  return mndwi
    .toFloat()
    .copyProperties(image, ['system:time_start']);
}

// ============================================================
// 2. LANDSAT CLOUD MASK
// ============================================================

function maskLandsat(image) {
  var qa = image.select('QA_PIXEL');
  var saturation = image.select('QA_RADSAT');

  var mask = qa.bitwiseAnd(1 << 0).eq(0)
    .and(qa.bitwiseAnd(1 << 1).eq(0))
    .and(qa.bitwiseAnd(1 << 2).eq(0))
    .and(qa.bitwiseAnd(1 << 3).eq(0))
    .and(qa.bitwiseAnd(1 << 4).eq(0))
    .and(qa.bitwiseAnd(1 << 5).eq(0))
    .and(saturation.eq(0));

  return image.updateMask(mask);
}


// ============================================================
// 3. MSAVI2 FUNCTION
// ============================================================

function calculateMSAVI(image, nirBand, redBand) {
  image = maskLandsat(image);

  var nir = image.select(nirBand)
    .multiply(0.0000275)
    .add(-0.2);

  var red = image.select(redBand)
    .multiply(0.0000275)
    .add(-0.2);

  var discriminant = nir.multiply(2)
    .add(1)
    .pow(2)
    .subtract(nir.subtract(red).multiply(8))
    .max(0);

  var msavi = nir.multiply(2)
    .add(1)
    .subtract(discriminant.sqrt())
    .divide(2)
    .rename('MSAVI');

  return msavi
    .updateMask(msavi.gte(-1).and(msavi.lte(1)))
    .toFloat()
    .copyProperties(image, ['system:time_start']);
}

// ============================================================
// 7. THEIL–SEN SLOPE
// Median of pairwise annual slopes
// Units: MSAVI units/year
// ============================================================

var pairwiseSlopes = [];

for (var i = 0; i < years.length - 1; i++) {
  for (var j = i + 1; j < years.length; j++) {

    var imageI = ee.Image(annualImages[i]);
    var imageJ = ee.Image(annualImages[j]);

    var slope = imageJ.subtract(imageI)
      .divide(years[j] - years[i])

// ============================================================

var pairwiseSigns = [];

for (var a = 0; a < years.length - 1; a++) {

    var imageB = ee.Image(annualImages[b]);

// Approximate variance without tie correction

  .multiply(n.subtract(1))
  .multiply(n.multiply(2).add(5))
  .divide(18);

  .where(mkS.gt(0), mkS.subtract(1))
  .where(mkS.lt(0), mkS.add(1));

var mkZ = numerator
  .divide(varianceS.sqrt())
  .rename('MK_Z')
  .updateMask(validMask)
  .clip(roi);

// Normal approximation
// ============================================================

  .subtract(
    mkZ.abs().divide(Math.sqrt(2)).erf()
  )
  .clamp(0, 1)
  .rename('MK_p')
  .updateMask(validMask)
  .clip(roi);

  .where(senSlope.lt(0).and(mkP.lt(ALPHA)), 2)
  .where(senSlope.gt(0).and(mkP.gte(ALPHA)), 3)
  .where(senSlope.lt(0).and(mkP.gte(ALPHA)), 4)
  .rename('Trend_Class')
  .updateMask(validMask)
  .clip(roi)
  .toByte();

var significance = mkP.lt(ALPHA)

  .updateMask(validMask)

/***************************************************************
 MODIS NDVI TREND ANALYSIS — SUNDARBANS
 Period: 2001–2020
 Dataset: MODIS/061/MOD13Q1
 Spatial resolution: 250 m
 Temporal resolution: 16 days

 Outputs:
 1. Annual NDVI stack
 2. Theil–Sen slope (NDVI units/year)
 3. Mann–Kendall S statistic
 4. Mann–Kendall Z statistic
 5. Mann–Kendall p-value
 6. Trend classification
 7. Significance mask
 8. Valid annual observation count
 9. Combined trend result
****************************************************************/

// ============================================================
// 1. STUDY AREA
// ============================================================
// Import your Sundarbans boundary as "roi" in the Imports panel.

Map.centerObject(roi, 8);
Map.addLayer(roi, {color: 'black'}, 'Study Area', false);

// ============================================================
// 2. PARAMETERS
// ============================================================

var START_YEAR = 2001;
var END_YEAR = 2020;
var ALPHA = 0.05;
var MIN_OBS = 15;

var SCALE = 250;
var CRS = 'EPSG:32645';
var EXPORT_FOLDER = 'NDVI_Trend';

var years = ee.List.sequence(START_YEAR, END_YEAR);

// ============================================================
// 3. MODIS NDVI DATA AND QUALITY MASK
// ============================================================

var modis = ee.ImageCollection('MODIS/061/MOD13Q1')
  .filterBounds(roi)
  .filterDate(
    ee.Date.fromYMD(START_YEAR, 1, 1),
    ee.Date.fromYMD(END_YEAR + 1, 1, 1)
  )
  .select(['NDVI', 'SummaryQA']);

print('MODIS image collection:', modis);
print('Number of MODIS images:', modis.size());

// Retain good-quality NDVI observations.
// SummaryQA = 0 indicates good quality.

function prepareNDVI(image) {
  var qa = image.select('SummaryQA');
  var ndvi = image.select('NDVI')
    .multiply(0.0001)
    .rename('NDVI');

  return ndvi
    .updateMask(qa.eq(0))
    .copyProperties(image, ['system:time_start']);
}

var ndviCollection = modis.map(prepareNDVI);

// ============================================================
// 4. ANNUAL MEAN NDVI COMPOSITES
// ============================================================

var annualImages = years.map(function(year) {
  year = ee.Number(year);

  var start = ee.Date.fromYMD(year, 1, 1);
  var end = start.advance(1, 'year');

  var yearly = ndviCollection.filterDate(start, end);

  var annualNDVI = yearly.mean()
    .rename('NDVI')
    .clip(roi)
    .set('year', year)
    .set('system:time_start', start.millis())
    .set('image_count', yearly.size());

  return annualNDVI;
});

var annualCollection = ee.ImageCollection.fromImages(annualImages);

print('Annual NDVI collection:', annualCollection);
print('Annual composite count:', annualCollection.size());

// Count valid annual observations at each pixel.

var validYearCount = annualCollection
  .count()
  .rename('Valid_Year_Count')
  .clip(roi);

Map.addLayer(
  validYearCount,
  {
    min: 0,
    max: 20,
    palette: ['white', 'yellow', 'green', 'darkgreen']
  },
  'Valid Annual Observations',
  false
);

// ============================================================
// 5. CREATE ANNUAL NDVI STACK
// ============================================================

var annualImageList = annualCollection.toList(20);

var bandNames = years.map(function(year) {
  return ee.String('NDVI_').cat(ee.Number(year).format('%d'));
});

var annualStack = ee.ImageCollection.fromImages(
  years.map(function(year) {
    year = ee.Number(year);

    var image = ee.Image(
      annualCollection
        .filter(ee.Filter.eq('year', year))
        .first()
    );

    return image.rename(
      ee.String('NDVI_').cat(year.format('%d'))
    );
  })
).toBands();

// Remove the automatic numeric prefixes from toBands().
annualStack = annualStack.rename(bandNames).clip(roi);

print('Annual NDVI stack:', annualStack);
print('Annual NDVI band names:', annualStack.bandNames());

// ============================================================
// 6. THEIL–SEN SLOPE AND MANN–KENDALL S STATISTIC
// ============================================================

// Theil–Sen slope:
// Median of all pairwise annual NDVI slopes.
//
// Mann–Kendall S:
// Sum of signs of all pairwise differences.

var slopeImages = [];
var signImages = [];

for (var i = 0; i < 20; i++) {

  var imageI = ee.Image(annualImageList.get(i))
    .select('NDVI');

  var yearI = START_YEAR + i;

  for (var j = i + 1; j < 20; j++) {

    var imageJ = ee.Image(annualImageList.get(j))
      .select('NDVI');

    var yearJ = START_YEAR + j;

    var difference = imageJ.subtract(imageI);

    // Pairwise slope in NDVI units per year.
    var pairSlope = difference
      .divide(yearJ - yearI)
      .rename('Slope');

    slopeImages.push(pairSlope);

    // Sign of pairwise difference:
    // +1 = increase; -1 = decrease; 0 = equal.
    var pairSign = difference.gt(0)
      .subtract(difference.lt(0))
      .rename('Sign');

    signImages.push(pairSign);
  }
}

var slopeCollection = ee.ImageCollection.fromImages(slopeImages);
var signCollection = ee.ImageCollection.fromImages(signImages);

// Theil–Sen median slope.
var senSlope = slopeCollection
  .median()
  .rename('Theil_Sen_Slope');

// Mann–Kendall S statistic.
var mkS = signCollection
  .sum()
  .rename('MK_S');

// Apply the minimum-observation mask.
var adequateData = validYearCount.gte(MIN_OBS);

senSlope = senSlope.updateMask(adequateData);
mkS = mkS.updateMask(adequateData);

// ============================================================
// 7. MANN–KENDALL Z STATISTIC
// ============================================================

// No-ties variance approximation:
// Var(S) = n(n-1)(2n+5)/18
//
// This version uses the valid annual observation count
// at each pixel. It does not correct for tied observations.

var n = validYearCount.toFloat();

var varianceS = n
  .multiply(n.subtract(1))
  .multiply(n.multiply(2).add(5))
  .divide(18);

var standardDeviationS = varianceS.sqrt();

// Continuity-corrected Z statistic.
var mkZ = ee.Image(0)
  .where(mkS.gt(0), mkS.subtract(1))
  .where(mkS.lt(0), mkS.add(1))
  .divide(standardDeviationS)
  .rename('MK_Z')
  .updateMask(adequateData);

// ============================================================
// 8. APPROXIMATE TWO-TAILED MANN–KENDALL P-VALUE
// ============================================================

// Approximation of the standard normal upper-tail probability.
// p = 2 * P(Z > |observed Z|)

var absZ = mkZ.abs();

var t = ee.Image(1).divide(
  ee.Image(1).add(absZ.multiply(0.2316419))
);

var t2 = t.pow(2);
var t3 = t.pow(3);
var t4 = t.pow(4);
var t5 = t.pow(5);

var polynomial = t.multiply(0.319381530)
  .subtract(t2.multiply(0.356563782))
  .add(t3.multiply(1.781477937))
  .subtract(t4.multiply(1.821255978))
  .add(t5.multiply(1.330274429));

var normalDensity = absZ.pow(2)
  .multiply(-0.5)
  .exp()
  .multiply(0.3989422804);

var mkP = normalDensity
  .multiply(polynomial)
  .multiply(2)
  .clamp(0, 1)
  .rename('MK_p_value')
  .updateMask(adequateData);

// ============================================================
// 9. TREND CLASSIFICATION
// ============================================================

// Classes:
// 0 = Zero slope
// 1 = Significant increase (slope > 0 and p < 0.05)
// 2 = Significant decrease (slope < 0 and p < 0.05)
// 3 = Non-significant increase
// 4 = Non-significant decrease

var trendClass = ee.Image(0)
  .where(senSlope.gt(0).and(mkP.lt(ALPHA)), 1)
  .where(senSlope.lt(0).and(mkP.lt(ALPHA)), 2)
  .where(senSlope.gt(0).and(mkP.gte(ALPHA)), 3)
  .where(senSlope.lt(0).and(mkP.gte(ALPHA)), 4)
  .rename('Trend_Class')
  .updateMask(adequateData)
  .clip(roi);

// Significance mask:
// 1 = statistically significant
// 0 = not statistically significant

var significanceMask = mkP.lt(ALPHA)
  .rename('Significance_Mask')
  .updateMask(adequateData)
  .clip(roi);

// ============================================================
// 10. COMBINED TREND RESULT
// ============================================================

var combinedResult = senSlope
  .addBands(mkS)
  .addBands(mkZ)
  .addBands(mkP)
  .addBands(trendClass)
  .addBands(significanceMask)
  .addBands(validYearCount)
  .clip(roi);

print('Combined NDVI trend result:', combinedResult);
print('Combined output bands:', combinedResult.bandNames());

  {
    min: -0.02,

  'NDVI Theil–Sen Slope'

Map.addLayer(

  {
    min: 0,
    max: 0.1,
    palette: ['darkgreen', 'yellow', 'red']
  },
  'Mann–Kendall P Value',
  false

Map.addLayer(
  trendClass,
  {
    min: 0,
    max: 4,
    palette: [
      '808080', // 0: zero slope
      '006400', // 1: significant increase
      '8B0000', // 2: significant decrease
      '90EE90', // 3: non-significant increase

  },
  'NDVI Trend Classes'

Map.addLayer(

  {
    min: 0,

  },

PTC

/***************************************************************
 MODIS PERCENT TREE COVER (PTC) TREND ANALYSIS
 Period: 2001–2020
 Dataset: MODIS/061/MOD44B
 Nominal resolution: 250 m

 Outputs:
 1. Annual mean PTC
 2. Theil-Sen slope
 3. Mann-Kendall S
 4. Mann-Kendall Z
 5. Mann-Kendall p-value
 6. Trend classification
 7. Significance mask
 8. Valid-year count

 Required imported geometry: roi
***************************************************************/

// ============================================================
// 1. SETTINGS
// ============================================================

var START_YEAR = 2001;
var END_YEAR = 2020;

var ALPHA = 0.05;
var MIN_OBS = 15;

var SCALE = 250;
var DISPLAY_SCALE = 500;
var EXPORT_FOLDER = 'PTC_Trend';

var years = [];
for (var yr = START_YEAR; yr <= END_YEAR; yr++) {
  years.push(yr);
}

var nYears = years.length;

Map.centerObject(roi, 6);
Map.addLayer(roi, {color: 'black'}, 'Study Area', false);

print('Analysis period:', START_YEAR + '–' + END_YEAR);
print('Number of years:', nYears);


// ============================================================
// 2. LOAD MODIS PERCENT TREE COVER
// ============================================================

// MOD44B is an annual vegetation structure product.
// Percent_Tree_Cover represents percentage tree cover.

var modisPTC = ee.ImageCollection('MODIS/061/MOD44B')
  .filterBounds(roi)
  .filterDate('2001-01-01', '2021-01-01')
  .select('Percent_Tree_Cover');

print('MODIS PTC image collection:', modisPTC);
print('Available PTC images:', modisPTC.size());


// ============================================================
// 3. CREATE ANNUAL PTC IMAGES
// ============================================================

var annualImages = years.map(function(year) {

  var start = ee.Date.fromYMD(year, 1, 1);
  var end = start.advance(1, 'year');

  var annualImage = modisPTC
    .filterDate(start, end)
    .mean()
    .rename('PTC')
    .toFloat()
    .clip(roi);

  return annualImage
    .set('year', year)
    .set('system:time_start', start.millis());
});

var annualPTC = ee.ImageCollection.fromImages(annualImages);

print('Annual PTC collection:', annualPTC);


// ============================================================
// 4. VALID YEAR COUNT
// ============================================================

var validCount = annualPTC
  .count()
  .rename('Valid_Years')
  .clip(roi);

var validMask = validCount.gte(MIN_OBS);

Map.addLayer(
  validCount.reproject({
    crs: 'EPSG:4326',
    scale: DISPLAY_SCALE
  }),
  {
    min: MIN_OBS,
    max: nYears,
    palette: ['yellow', 'lightgreen', 'darkgreen']
  },
  'Valid Annual Observations',
  false
);


// ============================================================
// 5. THEIL-SEN SLOPE
// Median of pairwise slopes
// Units: percentage points of tree cover per year
// ============================================================

var pairwiseSlopes = [];

for (var i = 0; i < nYears - 1; i++) {
  for (var j = i + 1; j < nYears; j++) {

    var imageI = ee.Image(annualImages[i]).select('PTC');
    var imageJ = ee.Image(annualImages[j]).select('PTC');

    var slope = imageJ.subtract(imageI)
      .divide(years[j] - years[i])
      .rename('Slope')
      .toFloat();

    pairwiseSlopes.push(slope);
  }
}

var senSlope = ee.ImageCollection
  .fromImages(pairwiseSlopes)
  .median()
  .rename('Theil_Sen_Slope')
  .updateMask(validMask)
  .clip(roi);


// ============================================================
// 6. MANN-KENDALL S STATISTIC
// S = sum of sign(PTCj - PTCi)
// ============================================================

var pairwiseSigns = [];

for (var a = 0; a < nYears - 1; a++) {
  for (var b = a + 1; b < nYears; b++) {

    var imgA = ee.Image(annualImages[a]).select('PTC');
    var imgB = ee.Image(annualImages[b]).select('PTC');

    var difference = imgB.subtract(imgA);

    var signImage = difference.gt(0)
      .subtract(difference.lt(0))
      .rename('Sign')
      .toFloat();

    pairwiseSigns.push(signImage);
  }
}

var mkS = ee.ImageCollection
  .fromImages(pairwiseSigns)
  .sum()
  .rename('MK_S')
  .updateMask(validMask)
  .clip(roi);


// ============================================================
// 7. MANN-KENDALL Z STATISTIC
// Approximate variance without tie correction
// ============================================================

var n = validCount.toFloat();

var varianceS = n
  .multiply(n.subtract(1))
  .multiply(n.multiply(2).add(5))
  .divide(18);

var zNumerator = ee.Image.constant(0)
  .where(mkS.gt(0), mkS.subtract(1))
  .where(mkS.lt(0), mkS.add(1));

var mkZ = zNumerator
  .divide(varianceS.sqrt())
  .rename('MK_Z')
  .updateMask(validMask)
  .clip(roi);


// ============================================================
// 8. MANN-KENDALL TWO-TAILED P-VALUE
// Normal approximation
// ============================================================

var mkP = ee.Image.constant(1)
  .subtract(
    mkZ.abs().divide(Math.sqrt(2)).erf()
  )
  .clamp(0, 1)
  .rename('MK_p_value')
  .updateMask(validMask)
  .clip(roi);


// ============================================================
// 9. TREND CLASSIFICATION
//
// 0 = zero slope
// 1 = significant increase
// 2 = significant decrease
// 3 = non-significant increase
// 4 = non-significant decrease
// ============================================================

var trendClass = ee.Image.constant(0)
  .where(senSlope.gt(0).and(mkP.lt(ALPHA)), 1)
  .where(senSlope.lt(0).and(mkP.lt(ALPHA)), 2)
  .where(senSlope.gt(0).and(mkP.gte(ALPHA)), 3)
  .where(senSlope.lt(0).and(mkP.gte(ALPHA)), 4)
  .rename('Trend_Class')
  .updateMask(validMask)
  .clip(roi)
  .toByte();

var significanceMask = mkP.lt(ALPHA)
  .rename('Significance_Mask')
  .updateMask(validMask)
  .clip(roi)
  .toByte();


// ============================================================
// 10. MAP DISPLAY
// Coarser visualization scale reduces tile-rendering load.
// ============================================================

Map.addLayer(
  senSlope.reproject({
    crs: 'EPSG:4326',
    scale: DISPLAY_SCALE
  }),
  {
    min: -1,
    max: 1,
    palette: ['blue', 'white', 'green']
  },
  'PTC Theil-Sen Slope'
);

Map.addLayer(
  trendClass.reproject({
    crs: 'EPSG:4326',
    scale: DISPLAY_SCALE
  }),
  {
    min: 0,
    max: 4,
    palette: [
      'bdbdbd',
      '238b45',
      '2166ac',
      'a1d99b',
      '9ecae1'
    ]
  },
  'PTC Trend Classes',
  false
);

Map.addLayer(
  mkP.reproject({
    crs: 'EPSG:4326',
    scale: DISPLAY_SCALE
  }),
  {
    min: 0,
    max: 0.1,
    palette: ['green', 'yellow', 'red']
  },
  'PTC Mann-Kendall p-value',
  false
);

Map.addLayer(
  significanceMask.reproject({
    crs: 'EPSG:4326',
    scale: DISPLAY_SCALE
  }),
  {
    min: 0,
    max: 1,
    palette: ['white', 'black']
  },
  'PTC Significance Mask',
  false
);


// ============================================================
// 11. EXPORT FUNCTION
// ============================================================

function exportResult(image, name) {

  Export.image.toDrive({
    image: image,
    description: name + '_2001_2020',
    folder: EXPORT_FOLDER,
    fileNamePrefix: name + '_2001_2020',
    region: roi,
    scale: SCALE,
    maxPixels: 1e13,
    fileFormat: 'GeoTIFF'
  });
}


// ============================================================
// 12. EXPORT INDIVIDUAL RESULTS
// ============================================================

exportResult(senSlope, 'PTC_Theil_Sen_Slope');
exportResult(mkS, 'PTC_MK_S');
exportResult(mkZ, 'PTC_MK_Z');
exportResult(mkP, 'PTC_MK_p_value');
exportResult(trendClass, 'PTC_Trend_Class');
exportResult(significanceMask, 'PTC_Significance_Mask');
exportResult(validCount, 'PTC_Valid_Year_Count');

print('PTC trend analysis prepared.');
print('Open the Tasks tab and run the exports.');
